      expected_target_version,target_snapshot_hash,
      model_config_id,model_config_version,resolved_config_hash,requested_provider,requested_model,
      parameters_snapshot_json,max_input_tokens,max_output_tokens,max_attempts,
      prompt_id,prompt_version,prompt_hash,provider_envelope_version,provider_envelope_hash,
      input_schema_version,output_schema_version,policy_version,input_context_json,input_hash,
      execution_environment,budget_policy_version,run_cost_limit_microusd,

      ${target === "revision" ? core.revisionId : null},${target === "revision" ? null : "en"},
      ${options.expectedTargetVersion ?? 1},${hash64("e")},
      ${core.defaultConfigId},1,${hash64("f")},'deepseek','deepseek-v4-flash',
      ${sql.json({})},16000,4000,3,'product-description-draft',1,${hash64("0")},1,${hash64("1")},

      ${staging ? 5000000 : 0},${staging ? 50000000 : 0},${staging ? 100000000 : 0},
      ${staging ? 20000 : 0},${sql.json(staging ? { source: "synthetic-review" } : {})}

}

async function enqueueIdempotent(
  sql: Sql,
  core: Core,
  key: string,
  fingerprint: string,
): Promise<{ kind: "inserted" | "replay" | "conflict"; id: string }> {
  return sql.begin(async (tx) => {
    const proposedId = randomUUID();
    const inserted = await tx<{ id: string }[]>`
      insert into ai_runs(
        id,use_case,requested_by_user_id,idempotency_key,request_fingerprint,
        target_type,target_product_id,target_locale,expected_target_version,target_snapshot_hash,
        model_config_id,model_config_version,resolved_config_hash,requested_provider,requested_model,
        parameters_snapshot_json,max_input_tokens,max_output_tokens,max_attempts,
        prompt_id,prompt_version,prompt_hash,provider_envelope_version,provider_envelope_hash,
        input_schema_version,output_schema_version,policy_version,input_context_json,input_hash,
        execution_environment,budget_policy_version,run_cost_limit_microusd,
        daily_hard_limit_microusd,monthly_warning_limit_microusd,monthly_hard_limit_microusd,
        estimated_max_cost_microusd,pricing_snapshot_json
      ) values (
        ${proposedId},'product_description_draft',${core.actorId},${key},${fingerprint},
        'product_draft',${core.productId},'en',1,${hash64("e")},
        ${core.defaultConfigId},1,${hash64("f")},'deepseek','deepseek-v4-flash',
        ${tx.json({})},16000,4000,3,'product-description-draft',1,${hash64("0")},1,${hash64("1")},
        1,1,'stage4a-v1',${tx.json({ synthetic: true })},${hash64("2")},
        'staging','stage4a-staging-v1',20000,5000000,50000000,100000000,20000,
        ${tx.json({ source: "synthetic-review" })}
      ) on conflict (idempotency_key) do nothing returning id
    `;
    if (inserted[0]) {
      await tx`
        insert into audit_logs(actor_user_id,action,entity_type,entity_id,after_summary)
        values (${core.actorId},'ai.run.enqueued','ai_run',${inserted[0].id},${tx.json({ fingerprint })})
      `;
      return { kind: "inserted", id: inserted[0].id };
    }
    const [existing] = await tx<{ id: string; request_fingerprint: string }[]>`
      select id,request_fingerprint from ai_runs
      where idempotency_key=${key} and requested_by_user_id=${core.actorId}
    `;
    assert.ok(existing);
    return existing.request_fingerprint === fingerprint
      ? { kind: "replay", id: existing.id }
      : { kind: "conflict", id: existing.id };
  });
}

type ClaimResult = { kind: "claimed"; id: string; token: string } | { kind: "busy" | "empty" | "concurrency" | "budget" };

async function claimOnce(sql: Sql): Promise<ClaimResult> {
  return sql.begin(async (tx) => {
    const [lock] = await tx<{ acquired: boolean }[]>`
      select pg_try_advisory_xact_lock(${claimLock[0]}::integer,${claimLock[1]}::integer) as acquired
    `;
    if (!lock?.acquired) return { kind: "busy" };
    const [active] = await tx<{ count: number }[]>`
      select count(*)::int as count from ai_runs
      where execution_environment='staging' and status='processing'
        and lease_expires_at > clock_timestamp()
    `;
    if ((active?.count ?? 0) >= 2) return { kind: "concurrency" };
    const [run] = await tx<{ id: string; run_cost_limit_microusd: number; cost_accounting_state: string; budget_reserved_cost_microusd: number }[]>`
      select id,run_cost_limit_microusd,budget_reserved_cost_microusd,cost_accounting_state
      from ai_runs
      where execution_environment='staging' and status='pending' and next_attempt_at <= clock_timestamp()
      order by next_attempt_at,queued_at,id
      for update skip locked limit 1
    `;
    if (!run) return { kind: "empty" };
    const [period] = await tx<{ day: string; month: string }[]>`
      select (transaction_timestamp() at time zone 'Asia/Shanghai')::date::text as day,
             date_trunc('month',transaction_timestamp() at time zone 'Asia/Shanghai')::date::text as month
    `;
    assert.ok(period);
    const [totals] = await tx<{ day_total: number; month_total: number }[]>`
      select
        coalesce(sum(budget_accounted_cost_microusd+budget_reserved_cost_microusd)
          filter (where budget_charge_day=${period.day}::date),0)::bigint::text::numeric::int as day_total,
        coalesce(sum(budget_accounted_cost_microusd+budget_reserved_cost_microusd)
          filter (where budget_charge_month=${period.month}::date),0)::bigint::text::numeric::int as month_total
      from ai_runs where execution_environment='staging'
    `;
    const additional = run.cost_accounting_state === "preflight" ? Number(run.run_cost_limit_microusd) : 0;
    if ((totals?.day_total ?? 0) + additional > 5000000 || (totals?.month_total ?? 0) + additional > 100000000) {
      return { kind: "budget" };
    }
    const token = randomUUID();
    const updated = await tx<{ id: string }[]>`
      update ai_runs set status='processing',retry_state='none',attempt_count=attempt_count+1,
        next_attempt_at=null,lease_owner='synthetic-claim-worker',lease_token=${token},
        lease_acquired_at=clock_timestamp(),lease_expires_at=clock_timestamp()+interval '10 minutes',
        budget_charge_day=coalesce(budget_charge_day,${period.day}::date),
        budget_charge_month=coalesce(budget_charge_month,${period.month}::date),
        budget_reserved_cost_microusd=case when cost_accounting_state='preflight' then run_cost_limit_microusd else budget_reserved_cost_microusd end,
        cost_accounting_state='reserved',failure_code=null,failure_detail=null,
        state_version=state_version+1,updated_at=clock_timestamp()
      where id=${run.id} and status='pending'
      returning id
    `;
    assert.equal(updated.length, 1);
    return { kind: "claimed", id: run.id, token };
  });
}

async function claimWithRetry(sql: Sql): Promise<ClaimResult> {
  for (let attempt = 0; attempt < 50; attempt += 1) {
    const result = await claimOnce(sql);
    if (result.kind !== "busy") return result;
    await new Promise((resolvePromise) => setTimeout(resolvePromise, 5));
  }
  return { kind: "busy" };
}

async function dispatch(sql: Sql, id: string, token: string, expectedVersion?: number): Promise<number> {
  const rows = await sql<{ state_version: number }[]>`
    update ai_runs set active_attempt_dispatched_at=statement_timestamp(),
      provider_dispatched_at=coalesce(provider_dispatched_at,statement_timestamp()),
      actual_provider=requested_provider,state_version=state_version+1,updated_at=clock_timestamp()
    where id=${id} and status='processing' and lease_token=${token}
      and (${expectedVersion ?? null}::bigint is null or state_version=${expectedVersion ?? null})
      and lease_expires_at>clock_timestamp()
    returning state_version
  `;
  return rows.length;
}

async function scheduleRetry(sql: Sql, id: string, token: string, debit = 5000): Promise<void> {
  const updated = await sql<{ id: string }[]>`
    update ai_runs set status='pending',retry_state='scheduled',next_attempt_at=clock_timestamp(),
      failure_code='synthetic_transient',provider_response_status='timeout',generated_at=clock_timestamp(),
      attempt_history_json=attempt_history_json || ${sql.json([{ attempt: 1, outcome: "retry_scheduled", synthetic: true }])}::jsonb,
      budget_accounted_cost_microusd=${debit},budget_reserved_cost_microusd=run_cost_limit_microusd-${debit},
      actual_cost_complete=false,active_attempt_dispatched_at=null,
      lease_owner=null,lease_token=null,lease_acquired_at=null,lease_expires_at=null,
      state_version=state_version+1,updated_at=clock_timestamp()
    where id=${id} and status='processing' and lease_token=${token}
    returning id
  `;
  assert.equal(updated.length, 1);
}

async function failFinal(sql: Sql, id: string, accounted = 0): Promise<void> {
  await sql`
    update ai_runs set status='failed',retry_state='not_retryable',next_attempt_at=null,
      failure_code='synthetic_failure',completed_at=clock_timestamp(),
      lease_owner=null,lease_token=null,lease_acquired_at=null,lease_expires_at=null,
      active_attempt_dispatched_at=null,budget_accounted_cost_microusd=${accounted},
      budget_reserved_cost_microusd=0,cost_accounting_state='final',
      state_version=state_version+1,updated_at=clock_timestamp()
    where id=${id}
  `;
}

async function clearScenario(sql: Sql): Promise<void> {
  await sql`delete from ai_runs`;
  await sql`delete from audit_logs where action like 'ai.%' or action like 'synthetic.%'`;
}

async function lockHealth(sql: Sql): Promise<JsonRecord> {
  const [row] = await sql<JsonRecord[]>`
    select
      (select count(*)::int from pg_stat_activity where datname=current_database() and state='idle in transaction') as idle_in_transaction,
      (select count(*)::int from pg_locks l join pg_database d on d.oid=l.database where d.datname=current_database() and l.locktype='advisory') as advisory_locks,
      (select count(*)::int from pg_locks l join pg_database d on d.oid=l.database where d.datname=current_database() and l.locktype='tuple') as tuple_locks
  `;
  return row ?? {};
}

async function runIdempotency(sql: Sql, core: Core): Promise<JsonRecord> {
  await clearScenario(sql);
  const sameKey = randomUUID();
  const same = await Promise.all([
    enqueueIdempotent(sql, core, sameKey, hash64("3")),
    enqueueIdempotent(sql, core, sameKey, hash64("3")),
  ]);
  assert.equal(new Set(same.map((item) => item.id)).size, 1);
  assert.deepEqual(same.map((item) => item.kind).sort(), ["inserted", "replay"]);
  const [sameCounts] = await sql<JsonRecord[]>`
    select count(*)::int as runs,
      (select count(*)::int from audit_logs where action='ai.run.enqueued') as audits
    from ai_runs where idempotency_key=${sameKey}
  `;
  assert.deepEqual(sameCounts, { runs: 1, audits: 1 });

  await clearScenario(sql);
  const conflictKey = randomUUID();
  const conflicting = await Promise.all([
    enqueueIdempotent(sql, core, conflictKey, hash64("4")),
    enqueueIdempotent(sql, core, conflictKey, hash64("5")),
  ]);
  assert.equal(new Set(conflicting.map((item) => item.id)).size, 1);
  assert.deepEqual(conflicting.map((item) => item.kind).sort(), ["conflict", "inserted"]);
  const [conflictCounts] = await sql<JsonRecord[]>`
    select count(*)::int as runs,
      (select count(*)::int from audit_logs where action='ai.run.enqueued') as audits
    from ai_runs where idempotency_key=${conflictKey}
  `;
  assert.deepEqual(conflictCounts, { runs: 1, audits: 1 });
  return { sameKey: { callers: same, counts: sameCounts }, differentFingerprint: { callers: conflicting, counts: conflictCounts, priorPayloadReturned: false } };
}

async function runSkipLockedAndClaim(sql: Sql, core: Core): Promise<JsonRecord> {
  await clearScenario(sql);
  const first = await insertPending(sql, core);
  const second = await insertPending(sql, core);
  let release!: () => void;
  let locked!: () => void;
  const releasePromise = new Promise<void>((resolvePromise) => { release = resolvePromise; });
  const lockedPromise = new Promise<void>((resolvePromise) => { locked = resolvePromise; });
  const blocker = sql.begin(async (tx) => {
    const [row] = await tx<{ id: string }[]>`
      select id from ai_runs where id in (${first},${second}) order by queued_at,id for update limit 1
    `;
    locked();
    await releasePromise;
    return row?.id;
  });
  await lockedPromise;
  const [skipped] = await sql<{ id: string }[]>`
    select id from ai_runs where id in (${first},${second}) order by queued_at,id for update skip locked limit 1
  `;
  release();
  const held = await blocker;
  assert.ok(held && skipped && held !== skipped.id);

  await clearScenario(sql);
  const one = await insertPending(sql, core);
  const oneClaims = await Promise.all([claimWithRetry(sql), claimWithRetry(sql)]);
  assert.equal(oneClaims.filter((item) => item.kind === "claimed").length, 1);
  const [oneState] = await sql<JsonRecord[]>`select id,status,attempt_count,lease_token is not null as has_token from ai_runs where id=${one}`;
  assert.deepEqual(oneState, { id: one, status: "processing", attempt_count: 1, has_token: true });

  await clearScenario(sql);
  const three = await Promise.all([insertPending(sql, core), insertPending(sql, core), insertPending(sql, core)]);
  const threeClaims = await Promise.all([claimWithRetry(sql), claimWithRetry(sql), claimWithRetry(sql)]);
  const [active] = await sql<{ count: number }[]>`
    select count(*)::int as count from ai_runs where status='processing' and lease_expires_at>clock_timestamp()
  `;
  assert.equal(active?.count, 2);
  assert.equal(threeClaims.filter((item) => item.kind === "claimed").length, 2);
  assert.ok(threeClaims.some((item) => item.kind === "concurrency"));
  return { skipLocked: { held, skipped: skipped.id }, singleRun: { run: one, results: oneClaims, state: oneState }, concurrencyTwo: { runs: three, results: threeClaims, active: active?.count } };
}

async function runRetryRecoveryDispatch(sql: Sql, core: Core): Promise<JsonRecord> {
  await clearScenario(sql);
  const run = await insertPending(sql, core);
  const firstClaim = await claimWithRetry(sql);

    current_undispatched: true,

    where id=${run}
  `;

  `;
  assert.equal(recoveredUndispatched?.status, "pending");
  assert.equal(recoveredUndispatched?.history_length, 2);

  await clearScenario(sql);
  const dispatchedCrash = await insertPending(sql, core);
  const crashClaim = await claimWithRetry(sql);
  assert.equal(crashClaim.kind, "claimed");
  if (crashClaim.kind !== "claimed") assert.fail();
  const wrongMarker = await dispatch(sql, dispatchedCrash, randomUUID());
  let providerCalls = 0;
  if (wrongMarker === 1) providerCalls += 1;
  assert.equal(providerCalls, 0);
  assert.equal(await dispatch(sql, dispatchedCrash, crashClaim.token), 1);
  providerCalls += 1;
  await sql`
    update ai_runs set queued_at=statement_timestamp()-interval '3 minutes',
      lease_acquired_at=statement_timestamp()-interval '2 minutes',
      provider_dispatched_at=statement_timestamp()-interval '90 seconds',
      active_attempt_dispatched_at=statement_timestamp()-interval '90 seconds',
      lease_expires_at=statement_timestamp()-interval '1 minute',updated_at=statement_timestamp()
    where id=${dispatchedCrash}
  `;
  await sql.begin(async (tx) => {
    const [lock] = await tx<{ acquired: boolean }[]>`select pg_try_advisory_xact_lock(${claimLock[0]},${claimLock[1]}) as acquired`;
    assert.equal(lock?.acquired, true);
    const updated = await tx<{ id: string }[]>`
      update ai_runs set status='pending',retry_state='scheduled',next_attempt_at=clock_timestamp(),
        failure_code='lease_expired_dispatched',provider_response_status='timeout',actual_cost_complete=false,

    `;
    assert.equal(updated.length, 1);
  });
  const [recoveredDispatched] = await sql<JsonRecord[]>`

  `;

}

async function switchDefault(sql: Sql, core: Core, targetId: string, expected: Map<string, number>): Promise<"switched" | "conflict"> {
  return sql.begin(async (tx) => {
    const rows = await tx<{ id: string; record_version: number }[]>`
      select id,record_version from ai_model_config
      where use_case='product_description_draft' order by id for update
    `;
    for (const row of rows) if (expected.get(row.id) !== Number(row.record_version)) return "conflict";
    await tx`
      update ai_model_config set enabled=false,is_default=false,record_version=record_version+1,
        updated_by_user_id=${core.actorId},updated_at=clock_timestamp()
      where use_case='product_description_draft' and enabled=true and is_default=true
    `;
    await tx`
      update ai_model_config set enabled=true,is_default=true,record_version=record_version+1,
        updated_by_user_id=${core.actorId},updated_at=clock_timestamp()
      where id=${targetId}
    `;
    await tx`
      insert into audit_logs(actor_user_id,action,entity_type,entity_id)
      values (${core.actorId},'ai.model_config.default_switched','ai_model_config',${targetId})
    `;
    return "switched";
  });
}

async function runDefaultSwitchAudit(sql: Sql, core: Core): Promise<JsonRecord> {
  await clearScenario(sql);
  const before = await configSignature(sql);
  const expected = new Map(before.map((row) => [String(row.id), Number(row.record_version)]));
  const results = await Promise.all([
    switchDefault(sql, core, core.alternateConfigId, expected),
    switchDefault(sql, core, core.thirdConfigId, expected),
  ]);
  assert.deepEqual(results.sort(), ["conflict", "switched"]);
  const [defaultCount] = await sql<{ count: number }[]>`
    select count(*)::int as count from ai_model_config where enabled and is_default and use_case='product_description_draft'
  `;
  assert.equal(defaultCount?.count, 1);
  const [auditCount] = await sql<{ count: number }[]>`
    select count(*)::int as count from audit_logs where action='ai.model_config.default_switched'
  `;
  assert.equal(auditCount?.count, 1);

  const signatureBeforeRollback = await configSignature(sql);
  await sql.unsafe(`
    create function cwt_s4a_review_fail_config_audit() returns trigger language plpgsql as $$
    begin
      if new.action='ai.model_config.default_switched' then raise exception 'synthetic required audit failure'; end if;
      return new;
    end $$;
    create trigger cwt_s4a_review_fail_config_audit before insert on audit_logs
    for each row execute function cwt_s4a_review_fail_config_audit();
  `);
  const currentDefault = signatureBeforeRollback.find((row) => row.enabled && row.is_default);
  assert.ok(currentDefault);
  const rollbackTarget = String(currentDefault.id) === core.alternateConfigId ? core.thirdConfigId : core.alternateConfigId;
  const rollbackExpected = new Map(signatureBeforeRollback.map((row) => [String(row.id), Number(row.record_version)]));
  let rollbackCode: string | undefined;
  try {
    await switchDefault(sql, core, rollbackTarget, rollbackExpected);
  } catch (error) {
    rollbackCode = (error as { code?: string }).code;
  }
  await sql.unsafe(`drop trigger cwt_s4a_review_fail_config_audit on audit_logs; drop function cwt_s4a_review_fail_config_audit();`);
  assert.equal(rollbackCode, "P0001");
  const signatureAfterRollback = await configSignature(sql);
  assert.deepEqual(signatureAfterRollback, signatureBeforeRollback);
  return { concurrentResults: results, enabledDefaultCount: defaultCount?.count, auditCount: auditCount?.count, auditFailureSqlstate: rollbackCode, rollbackPreservedSignature: true };
}

async function acceptDirect(sql: Sql, core: Core, runId: string, target: "product" | "content", expectedVersion: number): Promise<"accepted" | "conflict"> {
  return sql.begin(async (tx) => {
    const table = target === "product" ? "product_localizations" : "content_localizations";
    const idColumn = target === "product" ? "product_id" : "content_id";
    const targetId = target === "product" ? core.productId : core.contentId;
    const rows = await tx.unsafe<{ editor_document_version: number }[]>(
      `select editor_document_version from ${table} where ${idColumn}=$1 and locale='en' for update`,
      [targetId],
    );
    if (rows[0]?.editor_document_version !== expectedVersion) return "conflict";
    await tx.unsafe(
      `update ${table} set editor_document_version=editor_document_version+1 where ${idColumn}=$1 and locale='en' and editor_document_version=$2`,
      [targetId, expectedVersion],
    );
    const updated = await tx<{ id: string }[]>`
      update ai_runs set human_disposition='accepted',evaluated_by_user_id=${core.actorId},evaluated_at=clock_timestamp(),
        applied_target_version=${expectedVersion + 1},state_version=state_version+1,updated_at=clock_timestamp()
      where id=${runId} and status='draft_ready' and human_disposition='not_evaluated'
      returning id
    `;
    if (updated.length !== 1) throw new Error("synthetic run acceptance conflict");
    await tx`
      insert into audit_logs(actor_user_id,action,entity_type,entity_id)
      values (${core.actorId},'ai.run.accepted','ai_run',${runId})
    `;
    return "accepted";
  });
}

async function runAcceptanceFencing(sql: Sql, core: Core): Promise<JsonRecord> {
  await clearScenario(sql);
  await sql`update product_localizations set editor_document_version=1 where product_id=${core.productId} and locale='en'`;
  const run = await makeDraftReady(sql, core, "product", 1);
  const results = await Promise.all([
    acceptDirect(sql, core, run, "product", 1),
    acceptDirect(sql, core, run, "product", 1),
  ]);
  assert.deepEqual(results.sort(), ["accepted", "conflict"]);
  const [state] = await sql<JsonRecord[]>`
    select r.human_disposition,r.applied_target_version,
      p.editor_document_version,
      (select count(*)::int from audit_logs where action='ai.run.accepted' and entity_id=${run}) as audit_count
    from ai_runs r join product_localizations p on p.product_id=r.target_product_id and p.locale=r.target_locale
    where r.id=${run}
  `;
  assert.deepEqual(state, { human_disposition: "accepted", applied_target_version: 2, editor_document_version: 2, audit_count: 1 });

  await clearScenario(sql);
  await sql`update content_localizations set editor_document_version=1 where content_id=${core.contentId} and locale='en'`;
  const rollbackRun = await makeDraftReady(sql, core, "content", 1);
  await sql.unsafe(`
    create function cwt_s4a_review_fail_accept_audit() returns trigger language plpgsql as $$
    begin
      if new.action='ai.run.accepted' then raise exception 'synthetic required acceptance audit failure'; end if;
      return new;
    end $$;
    create trigger cwt_s4a_review_fail_accept_audit before insert on audit_logs
    for each row execute function cwt_s4a_review_fail_accept_audit();
  `);
  let rollbackCode: string | undefined;
  try {
    await acceptDirect(sql, core, rollbackRun, "content", 1);
  } catch (error) {
    rollbackCode = (error as { code?: string }).code;
  }
  await sql.unsafe(`drop trigger cwt_s4a_review_fail_accept_audit on audit_logs; drop function cwt_s4a_review_fail_accept_audit();`);
  assert.equal(rollbackCode, "P0001");
  const [rollbackState] = await sql<JsonRecord[]>`
    select r.human_disposition,r.applied_target_version,c.editor_document_version,
      (select count(*)::int from audit_logs where action='ai.run.accepted' and entity_id=${rollbackRun}) as audit_count
    from ai_runs r join content_localizations c on c.content_id=r.target_content_id and c.locale=r.target_locale
    where r.id=${rollbackRun}
  `;
  assert.deepEqual(rollbackState, { human_disposition: "not_evaluated", applied_target_version: null, editor_document_version: 1, audit_count: 0 });
  return { concurrentAcceptance: { results, state }, requiredAuditRollback: { sqlstate: rollbackCode, state: rollbackState } };
}

async function main(): Promise<void> {
  assert.ok(evidencePath);
  const adminUrl = validateAdminUrl();
  const admin = postgres(adminUrl.toString(), { max: 1, prepare: false, onnotice() {} });

  assertDbName(name);
  const result: JsonRecord = { database: name };
  try {
    await admin.unsafe(`create database "${name}"`);

    try { await migratePostgresWithEnumCompatibility(migrator, join(projectRoot, "drizzle")); }
    finally { await migrator.end(); }
    const sql = postgres(url, { max: 16, prepare: false, onnotice() {} });
    try {
      const [identity] = await sql<JsonRecord[]>`
        select version() as version,current_setting('server_version_num') as server_version_num,
          current_database() as database,current_user as role
      `;
      assert.equal(identity?.server_version_num, "180004");
      result.postgresql = identity;
      const core = await seedCore(sql);
      result.idempotency = await runIdempotency(sql, core);

  }

  process.stdout.write(`PASS contention/lifecycle/budget review: ${evidencePath}\n`);
}
