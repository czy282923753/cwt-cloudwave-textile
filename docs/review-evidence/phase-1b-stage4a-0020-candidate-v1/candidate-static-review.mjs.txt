import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { readFile, writeFile } from "node:fs/promises";

const root = process.cwd();
const head = "15bc6462d2e314f50ff238af70ad31fc6502c40f";
const parent = "31c0e405acfdd0d05200d0fb2531e897a541a2c4";
const branch = "codex/phase-1b-stage4a-0020-migration-candidate-v1";
const tag = "phase-1b-stage3-approved-2026-08-09";
const designPath = "docs/PHASE_1B_STAGE4A_PHASE_A_0020_AI_FOUNDATION_SCHEMA_DESIGN_V1_0.md";
const externalReviewPath = "/Users/calvin/.codex/worktrees/900f/CWT（CloudWave Textile）项目/docs/PHASE_1B_STAGE4A_PHASE_A_0020_INDEPENDENT_MIGRATION_REVIEW_V1_0.md";
const outputPath = process.env.CWT_REVIEW_EVIDENCE_PATH;
assert.ok(outputPath);

const expectedChanged = [
  designPath,
  "drizzle/0020_phase1b_ai_foundation.sql",
  "drizzle/meta/0020_snapshot.json",
  "drizzle/meta/_journal.json",
  "package.json",
  "scripts/verify-ai-foundation-candidate.ts",
  "src/db/schema/ai.integration.test.ts",
  "src/db/schema/ai.ts",
  "src/db/schema/index.ts",
].sort();

function git(...args) {
  return execFileSync("git", args, { cwd: root, encoding: "utf8" }).trim();
}

function gitBuffer(...args) {
  return execFileSync("git", args, { cwd: root });
}

function sha(buffer) {
  return createHash("sha256").update(buffer).digest("hex");
}

async function fileSha(path) {
  return sha(await readFile(path));
}

const changedStatus = git("diff", "--name-status", `${parent}..${head}`).split("\n").filter(Boolean);
const changed = changedStatus.map((line) => line.split("\t").at(-1)).sort();
assert.deepEqual(changed, expectedChanged);

const treeFiles = git("ls-tree", "-r", "--name-only", parent).split("\n");
const historical = treeFiles.filter((path) =>
  /^drizzle\/(?:00(?:0\d|1\d))_.+\.sql$/.test(path)
  || /^drizzle\/meta\/00(?:0\d|1\d)_snapshot\.json$/.test(path)
).sort();
assert.equal(historical.length, 40);

const historicalIdentity = historical.map((path) => {
  const parentBuffer = gitBuffer("show", `${parent}:${path}`);
  const candidateBuffer = gitBuffer("show", `${head}:${path}`);
  const parentSha256 = sha(parentBuffer);
  const candidateSha256 = sha(candidateBuffer);
  assert.equal(candidateSha256, parentSha256, path);
  return { path, parentSha256, candidateSha256, identical: true };
});

const candidateHashes = {};
for (const path of expectedChanged) candidateHashes[path] = await fileSha(path);

const journal = JSON.parse(await readFile("drizzle/meta/_journal.json", "utf8"));
const snapshot = JSON.parse(await readFile("drizzle/meta/0020_snapshot.json", "utf8"));
const candidateTrackedDiff = git("diff", "--name-only", head, "--", ...expectedChanged);
assert.equal(candidateTrackedDiff, "");

const implementationPaths = [
  "drizzle/0020_phase1b_ai_foundation.sql",
  "package.json",
  "scripts/verify-ai-foundation-candidate.ts",
  "src/db/schema/ai.ts",
  "src/db/schema/index.ts",
];
const implementationDiff = git("diff", `${parent}..${head}`, "--", ...implementationPaths);
const forbiddenMechanismPatterns = {
  providerNetworkCall: /\b(?:fetch|axios|openai|anthropic)\s*\(/gi,
  credentialEnvironment: /process\.env\.(?:DEEPSEEK|OPENAI|ANTHROPIC|AI_API_KEY)/gi,
  ragOrVectorDependency: /\b(?:pgvector|vector store|embedding client|retrieval client)\b/gi,
};
const forbiddenMechanismMatches = Object.fromEntries(Object.entries(forbiddenMechanismPatterns).map(([name, pattern]) => [name, [...implementationDiff.matchAll(pattern)].map((match) => match[0])]));
for (const matches of Object.values(forbiddenMechanismMatches)) assert.equal(matches.length, 0);

const result = {
  generatedAt: new Date().toISOString(),
  repository: root,
  git: {
    head: git("rev-parse", "HEAD"),
    parent: git("rev-parse", "HEAD^"),
    branchRef: git("rev-parse", `refs/heads/${branch}`),
    checkedOutBranch: git("branch", "--show-current"),
    detached: git("rev-parse", "--abbrev-ref", "HEAD") === "HEAD",
    tagObject: git("rev-parse", tag),
    tagPeeledCommit: git("rev-parse", `${tag}^{commit}`),
    statusShort: git("status", "--short"),
  },
  fixedInputs: {
    design: { path: designPath, sha256: await fileSha(designPath) },
    independentDesignReview: { path: externalReviewPath, sha256: await fileSha(externalReviewPath) },
  },
  candidate: {
    expectedChanged,
    changedStatus,
    changedFileCount: changed.length,
    hashes: candidateHashes,
    trackedCandidateFilesUnmodified: candidateTrackedDiff === "",
    diffStat: git("diff", "--stat", `${parent}..${head}`),
  },
  history: {
    artifactCount: historicalIdentity.length,
    allIdentical: historicalIdentity.every((entry) => entry.identical),
    artifacts: historicalIdentity,
  },
  drizzle: {
    journalEntryCount: journal.entries.length,
    latestJournalEntry: journal.entries.at(-1),
    snapshotId: snapshot.id,
    previousSnapshotId: snapshot.prevId,
    snapshotDialect: snapshot.dialect,
    aiSnapshotTables: Object.keys(snapshot.tables).filter((name) => name === "public.ai_model_config" || name === "public.ai_runs").sort(),
  },
  scope: {
    implementationPaths,
    forbiddenMechanismMatches,
    forbiddenMechanismsAbsent: Object.values(forbiddenMechanismMatches).every((matches) => matches.length === 0),
  },
};

assert.equal(result.git.head, head);
assert.equal(result.git.parent, parent);
assert.equal(result.git.branchRef, head);
assert.equal(result.git.tagPeeledCommit, parent);
assert.equal(result.fixedInputs.design.sha256, "db6ae44d3548e2c0c23ab2b95ee3550fefedb93224f878f5a9ab3070898b60a8");
assert.equal(result.fixedInputs.independentDesignReview.sha256, "fe17a42990f1b55fca89e1f038cede9c09aff3c418379ff8d4c54d882ff3e6b2");
assert.equal(result.drizzle.journalEntryCount, 21);
assert.equal(result.drizzle.latestJournalEntry.tag, "0020_phase1b_ai_foundation");
assert.deepEqual(result.drizzle.aiSnapshotTables, ["public.ai_model_config", "public.ai_runs"]);

await writeFile(outputPath, `${JSON.stringify(result, null, 2)}\n`, "utf8");
process.stdout.write(`Static Candidate review complete: ${changed.length} changed files; ${historicalIdentity.length} historical artifacts identical.\n`);
