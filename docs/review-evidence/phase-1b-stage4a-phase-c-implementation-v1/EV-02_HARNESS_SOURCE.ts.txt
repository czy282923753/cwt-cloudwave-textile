function deferred() {
  let resolve!: () => void;
  const promise = new Promise<void>((accept) => { resolve = accept; });
  return { promise, resolve };
}

async function waitFor(predicate: () => Promise<boolean>): Promise<void> {
  for (let attempt = 0; attempt < 1_000; attempt += 1) {
    if (await predicate()) return;
    await new Promise((resolve) => setTimeout(resolve, 10));
  }
  throw new Error("EV-02 condition timed out.");
}

}

async function twoSlotScenario() {
  const fixture = await resetAndSeed();
  await enqueue(fixture, 3);
  const release = deferred();
  let active = 0;
  let maximum = 0;
  const deferredProvider: TextAiProvider = {
    ...baseProvider,
    async generateText(input) {
      active += 1;
      maximum = Math.max(maximum, active);
      await release.promise;
      try { return await baseProvider.generateText(input); }
      finally { active -= 1; }
    },
  };
  const registry = createTextProviderRegistryV1([deferredProvider]);
  if (!registry.ok) throw new Error("Deferred registry failed.");
  const worker = createAiRunWorkerV1({
    database,
    trustedEnvironment: { appEnvironment: "test", processFeatureAiEnabled: true },
    providerRegistry: registry.value,
    promptLoader,
    pricingRegistry: localTestPricingPolicyRegistryV1,
    timing: { heartbeatIntervalMs: 15_000, lockRetryDelayMs: 1_000, idlePollMs: 10, gracefulShutdownMs: 1_000, postAbortPersistenceMs: 200 },
    workerId: "ev02-two-slot",
  });
  await worker.start();
  await waitFor(async () => active === 2);
  const blocked = await database.select().from(aiRuns);
  release.resolve();
  await waitFor(async () => (await database.select().from(aiRuns)).every((row) => row.status === "draft_ready"));
  await worker.stop("SIGTERM");
  const final = await database.select().from(aiRuns);
  return {
    maximumProviderCalls: maximum,
    blockedState: {
      processing: blocked.filter((row) => row.status === "processing").length,
      pending: blocked.filter((row) => row.status === "pending").length,
    },
    finalDraftReady: final.filter((row) => row.status === "draft_ready").length,
    providerCalls: requests.length,
  };
}

async function contentionScenario() {
  requests.length = 0;
  const fixture = await resetAndSeed();
  await enqueue(fixture, 1);
  const entered = deferred();
  let abortReason: unknown;
  const abortingProvider: TextAiProvider = {
    ...baseProvider,
    async generateText(input) {
      entered.resolve();
      await new Promise<void>((resolve) => {
        if (input.signal.aborted) resolve();
        else input.signal.addEventListener("abort", () => resolve(), { once: true });
      });
      abortReason = input.signal.reason;
      return { kind: "failure", responseStatus: "transport_error", failureCode: "transport", retryClass: "same_provider_transient", durationMs: 1 };
    },
  };
  const registry = createTextProviderRegistryV1([abortingProvider]);
  if (!registry.ok) throw new Error("Aborting registry failed.");
  const telemetry: string[] = [];
  const worker = createAiRunWorkerV1({
    database,
    trustedEnvironment: { appEnvironment: "test", processFeatureAiEnabled: true },
    providerRegistry: registry.value,
    promptLoader,
    pricingRegistry: localTestPricingPolicyRegistryV1,
    telemetry: { emit: (event) => { telemetry.push(event.eventName); } },
    timing: { heartbeatIntervalMs: 10, lockRetryDelayMs: 20, idlePollMs: 10, gracefulShutdownMs: 300, postAbortPersistenceMs: 150 },
    workerId: "ev02-contention",
  });
  await worker.start();
  await entered.promise;
  const lockClient = postgres(databaseUrl, { max: 1, prepare: false, onnotice: () => undefined });
  const held = deferred();
  const release = deferred();
  const advisory = CWT_AI_TEXT_CLAIM_BUDGET_ADVISORY_KEY_V1;
  const lockPromise = lockClient.begin(async (transaction) => {
    await transaction`select pg_advisory_xact_lock(${advisory[0]}, ${advisory[1]})`;
    held.resolve();
    await release.promise;
  });
  await held.promise;
  await worker.stop("SIGTERM");
  const [abandoned] = await database.select().from(aiRuns);
  release.resolve();
  await lockPromise;
  await lockClient.end();
  return {
    abortReason,
    heartbeatBusyEvents: telemetry.filter((event) => event === "ai_heartbeat_lock_busy").length,
    renewalUnavailableEvents: telemetry.filter((event) => event === "ai_lease_renewal_unavailable").length,
    durableStatus: abandoned?.status,
    candidateNull: abandoned?.candidateJson === null && abandoned?.candidateHash === null,
  };
}

  const twoSlot = await twoSlotScenario();

  const wallDurationMs = performance.now() - wallStarted;
  const cpu = process.cpuUsage(cpuBefore);
  const locks = await database.execute<{ count: number }>(sql`
    select count(*)::integer as count from pg_locks
    where locktype='advisory' and classid=1129792594 and objid=1
  `);
  const idle = await database.execute<{ count: number }>(sql`
    select count(*)::integer as count from pg_stat_activity
    where datname=current_database() and state='idle in transaction'
  `);
  const result = {
    schemaVersion: 1,
    synthetic: true,
    envelope: {
      targetVcpu: 2,
      targetMemoryBytes: 4_294_967_296,
      validationMode: "measured-demand-below-equivalent-capacity",
      hostAvailableParallelism: availableParallelism(),
    },
    node: process.version,
    twoSlot,
    contention,
    resource: {
      cpuUserMicroseconds: cpu.user,
      cpuSystemMicroseconds: cpu.system,
      rssBeforeBytes: rssBefore,
      rssAfterBytes: process.memoryUsage().rss,
      maxRssKilobytes: process.resourceUsage().maxRSS,
      wallDurationMs,
      averageVcpuDemand: (cpu.user + cpu.system) / (wallDurationMs * 1_000),
      eventLoopMeanMs: eventLoop.mean / 1e6,
      eventLoopP99Ms: eventLoop.percentile(99) / 1e6,
      eventLoopMaxMs: eventLoop.max / 1e6,
      dbPoolMax: 8,
    },
    health: { advisoryLocks: locks[0]?.count ?? -1, idleInTransaction: idle[0]?.count ?? -1 },
  };
  if (result.resource.averageVcpuDemand >= result.envelope.targetVcpu ||
    result.resource.maxRssKilobytes * 1_024 >= result.envelope.targetMemoryBytes ||
    twoSlot.maximumProviderCalls !== 2 || twoSlot.blockedState.processing !== 2 ||
    twoSlot.blockedState.pending !== 1 || twoSlot.finalDraftReady !== 3 ||
    contention.abortReason !== "lease_renewal_unavailable" || contention.heartbeatBusyEvents !== 5 ||
    contention.renewalUnavailableEvents !== 1 || contention.durableStatus !== "processing" ||
    !contention.candidateNull || result.health.advisoryLocks !== 0 || result.health.idleInTransaction !== 0) {
    throw new Error(`EV-02 assertion failed: ${JSON.stringify(result)}`);
  }
  process.stdout.write(`${JSON.stringify(result, null, 2)}\n`);
}
