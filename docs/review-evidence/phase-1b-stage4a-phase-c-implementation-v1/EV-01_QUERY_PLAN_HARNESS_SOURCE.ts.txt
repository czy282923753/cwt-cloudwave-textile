function validateAdminUrl(): URL {
  assert.ok(adminUrlText);
  const url = new URL(adminUrlText);
  assert.ok(["127.0.0.1", "localhost"].includes(url.hostname));
  assert.equal(url.pathname, "/cwt_review_admin");

function dbUrl(adminUrl: URL, name: string): string {
  const url = new URL(adminUrl);
  url.pathname = `/${name}`;
  return url.toString();
}

function summarizePlan(planValue: unknown): JsonRecord {
  const planArray = planValue as Array<{ Plan: JsonRecord; Settings?: JsonRecord; "Planning Time"?: number; "Execution Time"?: number }>;
  const document = planArray[0]!;
  const nodes: JsonRecord[] = [];
  const visit = (node: JsonRecord): void => {
    nodes.push({
      nodeType: node["Node Type"],
      relation: node["Relation Name"],
      index: node["Index Name"],
      scanDirection: node["Scan Direction"],
      sortMethod: node["Sort Method"],
      actualRows: node["Actual Rows"],
      actualLoops: node["Actual Loops"],
      sharedHitBlocks: node["Shared Hit Blocks"],
      sharedReadBlocks: node["Shared Read Blocks"],
    });
    for (const child of (node.Plans as JsonRecord[] | undefined) ?? []) visit(child);
  };
  visit(document.Plan);
  return {
    rootNode: document.Plan["Node Type"],
    nodes,
    settings: document.Settings ?? {},
    planningTimeMs: document["Planning Time"],
    executionTimeMs: document["Execution Time"],
    raw: planValue,
  };
}

async function explain(sql: Sql, query: string, parameters: unknown[] = []): Promise<JsonRecord> {
  const rows = await sql.unsafe<JsonRecord[]>(
    `explain (analyze, buffers, settings, format json) ${query}`,
    parameters,
  );
  return summarizePlan(rows[0]?.["QUERY PLAN"]);
}

function planUses(plan: JsonRecord, indexName: string): boolean {
  return (plan.nodes as JsonRecord[]).some((node) => node.index === indexName);
}

async function capturePlans(sql: Sql, fixture: JsonRecord): Promise<JsonRecord> {
  const sample = fixture.sample as { actor: string; product: string; content: string; revision: string; idempotency: string };
  const plans: Record<string, JsonRecord> = {};
  plans.resolveConfig = await explain(sql, `
    select id from ai_model_config
    where capability='text' and use_case='product_description_draft' and enabled and is_default
  `);
  plans.claimDue = await explain(sql, `
    select id from ai_runs
    where execution_environment='staging' and status='pending' and next_attempt_at<=clock_timestamp()
    order by next_attempt_at,queued_at,id for update skip locked limit 1
  `);
  plans.activeClock = await explain(sql, `
    select count(*) from ai_runs
    where execution_environment='staging' and status='processing' and lease_expires_at>clock_timestamp()
  `);
  plans.activeTransactionTimestamp = await explain(sql, `
    select count(*) from ai_runs
    where execution_environment='staging' and status='processing' and lease_expires_at>transaction_timestamp()
  `);
  plans.expiredLeases = await explain(sql, `
    select id from ai_runs
    where execution_environment='staging' and status='processing' and lease_expires_at<=clock_timestamp()
    order by lease_expires_at,id limit 50
  `);
  plans.budgetDay = await explain(sql, `
    select sum(budget_accounted_cost_microusd+budget_reserved_cost_microusd)
    from ai_runs where execution_environment='staging' and budget_charge_day=date '2025-01-01'
  `);
  plans.budgetMonth = await explain(sql, `
    select sum(budget_accounted_cost_microusd+budget_reserved_cost_microusd)
    from ai_runs where execution_environment='staging' and budget_charge_month=date '2025-01-01'
  `);
  plans.productHistory = await explain(sql, `
    select id from ai_runs where target_product_id=$1::uuid order by queued_at desc,id limit 50
  `, [sample.product]);
  plans.contentHistory = await explain(sql, `
    select id from ai_runs where target_content_id=$1::uuid order by queued_at desc,id limit 50
  `, [sample.content]);
  plans.revisionHistory = await explain(sql, `
    select id from ai_runs where target_revision_id=$1::uuid order by queued_at desc,id limit 50
  `, [sample.revision]);
  plans.requesterHistory = await explain(sql, `
    select id from ai_runs where requested_by_user_id=$1::uuid order by queued_at desc,id limit 50
  `, [sample.actor]);
  plans.adminList = await explain(sql, `
    select id from ai_runs where status='draft_ready' and use_case='product_description_draft'
    order by queued_at desc,id limit 50
  `);
  plans.idempotency = await explain(sql, `
    select id from ai_runs where idempotency_key=$1::uuid
  `, [sample.idempotency]);

  const expectations: Record<string, { index: string; passed: boolean }> = {
    resolveConfig: { index: "ai_model_config_enabled_default_unique", passed: planUses(plans.resolveConfig, "ai_model_config_enabled_default_unique") },
    claimDue: { index: "ai_runs_claimable_idx", passed: planUses(plans.claimDue, "ai_runs_claimable_idx") },
    activeClock: { index: "ai_runs_active_lease_idx", passed: planUses(plans.activeClock, "ai_runs_active_lease_idx") },
    activeTransactionTimestamp: { index: "ai_runs_active_lease_idx", passed: planUses(plans.activeTransactionTimestamp, "ai_runs_active_lease_idx") },
    expiredLeases: { index: "ai_runs_active_lease_idx", passed: planUses(plans.expiredLeases, "ai_runs_active_lease_idx") },
    budgetDay: { index: "ai_runs_budget_day_idx", passed: planUses(plans.budgetDay, "ai_runs_budget_day_idx") },
    budgetMonth: { index: "ai_runs_budget_month_idx", passed: planUses(plans.budgetMonth, "ai_runs_budget_month_idx") },
    productHistory: { index: "ai_runs_target_product_idx", passed: planUses(plans.productHistory, "ai_runs_target_product_idx") },
    contentHistory: { index: "ai_runs_target_content_idx", passed: planUses(plans.contentHistory, "ai_runs_target_content_idx") },
    revisionHistory: { index: "ai_runs_target_revision_idx", passed: planUses(plans.revisionHistory, "ai_runs_target_revision_idx") },
    requesterHistory: { index: "ai_runs_requester_history_idx", passed: planUses(plans.requesterHistory, "ai_runs_requester_history_idx") },
    adminList: { index: "ai_runs_admin_status_idx", passed: planUses(plans.adminList, "ai_runs_admin_status_idx") },
    idempotency: { index: "ai_runs_idempotency_key_unique", passed: planUses(plans.idempotency, "ai_runs_idempotency_key_unique") },
  };
  return {
    plans,
    expectations,
    passed: Object.values(expectations).filter((expectation) => expectation.passed).length,
    total: Object.keys(expectations).length,
    failed: Object.entries(expectations).filter(([, expectation]) => !expectation.passed).map(([name]) => name),
  };
}

async function main(): Promise<void> {
  assert.ok(evidencePath);
  const adminUrl = validateAdminUrl();
  const admin = postgres(adminUrl.toString(), { max: 1, prepare: false, onnotice() {} });
  const name = reviewDbName();
  assert.match(name, /^cwt_s4a_review_[a-z0-9_]+$/);
  const result: JsonRecord = { database: name };
  try {
    await admin.unsafe(`create database "${name}"`);
    const url = dbUrl(adminUrl, name);
    const migrator = postgres(url, { max: 1, prepare: false, onnotice() {} });
    try { await migratePostgresWithEnumCompatibility(migrator, join(projectRoot, "drizzle")); }
    finally { await migrator.end(); }
    const sql = postgres(url, { max: 4, prepare: false, onnotice() {} });
    try {
      const [identity] = await sql<JsonRecord[]>`
        select version() as version,current_setting('server_version_num') as server_version_num,
          current_database() as database,current_user as role
      `;
      assert.match(String(identity?.server_version_num), /^17\d{4}$/);
      result.postgresql = identity;
      result.fixture = await seedScale(sql);
      await sql.unsafe(`vacuum (analyze) ai_model_config`);
      await sql.unsafe(`vacuum (analyze) ai_runs`);
      const settings: JsonRecord = {};
      for (const setting of ["enable_seqscan", "enable_indexscan", "enable_bitmapscan", "enable_indexonlyscan", "random_page_cost", "seq_page_cost", "effective_cache_size", "work_mem", "max_parallel_workers_per_gather"]) {
        const [row] = await sql.unsafe<JsonRecord[]>(`show ${setting}`);
        settings[setting] = row ? Object.values(row)[0] : null;
      }
      result.plannerSettings = settings;
      assert.equal(settings.enable_seqscan, "on", "default planner must retain sequential scans");
      result.sizes = await sql<JsonRecord[]>`
        select c.relname as relation,c.reltuples::bigint::text as estimated_rows,
          pg_relation_size(c.oid)::text as relation_bytes,pg_total_relation_size(c.oid)::text as total_bytes
        from pg_class c join pg_namespace n on n.oid=c.relnamespace
        where n.nspname='public' and (c.relname in ('ai_model_config','ai_runs') or c.relname like 'ai_runs_%' or c.relname like 'ai_model_config_%')
        order by c.relname
      `;
      result.analyzeState = await sql<JsonRecord[]>`
        select relname,n_live_tup,last_vacuum,last_analyze
        from pg_stat_user_tables where relname in ('ai_model_config','ai_runs') order by relname
      `;
      result.queryPlans = await capturePlans(sql, result.fixture as JsonRecord);
      const [health] = await sql<JsonRecord[]>`
        select
          (select count(*)::int from pg_stat_activity where datname=current_database() and state='idle in transaction') as idle_in_transaction,
          (select count(*)::int from pg_locks l join pg_database d on d.oid=l.database where d.datname=current_database() and l.locktype='advisory') as advisory_locks,
          (select count(*)::int from pg_locks l join pg_database d on d.oid=l.database where d.datname=current_database() and l.locktype='tuple') as tuple_locks
      `;
      result.lockHealth = health;
      assert.deepEqual(health, { idle_in_transaction: 0, advisory_locks: 0, tuple_locks: 0 });
    } finally {
      await sql.end();
    }
  } finally {
    await admin.unsafe(`drop database if exists "${name}" with (force)`);
    const remaining = await admin<{ name: string }[]>`
      select datname as name from pg_database where datname like 'cwt_s4a_review_%' order by datname
    `;
    result.cleanup = { dropped: name, remainingReviewDatabases: remaining.map((row) => row.name) };
    await admin.end();
  }
  await writeFile(evidencePath, `${JSON.stringify(result, null, 2)}\n`, "utf8");
  const planResult = result.queryPlans as { passed: number; total: number; failed: string[] };
  process.stdout.write(`Scale query-plan review complete: ${planResult.passed}/${planResult.total}; failed=${planResult.failed.join(",") || "none"}\n`);
}
